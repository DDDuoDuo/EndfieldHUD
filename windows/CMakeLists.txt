cmake_minimum_required(VERSION 3.25)
project(EndfieldHUDWindows VERSION 0.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# Clean port of the current macOS application. The rejected Windows preview is
# not a dependency, build input, resource source or packageable application.
add_library(hud_scene core/scene.cpp)
target_include_directories(hud_scene PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(hud_motion core/motion.cpp)
target_include_directories(hud_motion PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(hud_data core/data/json.cpp core/data/file_io.cpp core/data/data_store.cpp core/data/notes_store.cpp)
target_include_directories(hud_data PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_compile_definitions(hud_data PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(hud_data PRIVATE winsqlite3)
else()
  find_package(SQLite3 REQUIRED)
  target_link_libraries(hud_data PRIVATE SQLite::SQLite3)
endif()
foreach(target hud_scene hud_motion hud_data)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
  foreach(name scene motion data)
    add_executable(${name}_tests tests/${name}_tests.cpp)
    target_link_libraries(${name}_tests PRIVATE hud_${name})
    if(MSVC)
      target_compile_options(${name}_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
    endif()
    add_test(NAME ${name}_contracts COMMAND ${name}_tests)
    set_tests_properties(${name}_contracts PROPERTIES TIMEOUT 60)
  endforeach()
endif()
