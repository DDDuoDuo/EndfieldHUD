cmake_minimum_required(VERSION 3.25)
project(EndfieldHUDWindows LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The feasibility target requires Windows x64. Other targets are unverified.")
endif()
if((CMAKE_GENERATOR_PLATFORM AND NOT CMAKE_GENERATOR_PLATFORM MATCHES "^x64(,.*)?$") OR
   CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64)$")
  message(FATAL_ERROR "Only the tested Windows x64 target is supported by this build.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
# The runtime needs only the bounded raw-DEFLATE decoder, not gzip I/O or a
# compressor. Pin the official source archive; include its license in packages.
FetchContent_Declare(zlib_source URL https://zlib.net/fossils/zlib-1.3.2.tar.gz
  URL_HASH SHA256=bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR endfield-no-subproject)
FetchContent_MakeAvailable(zlib_source)
add_library(ehud_inflate STATIC
  ${zlib_source_SOURCE_DIR}/adler32.c ${zlib_source_SOURCE_DIR}/crc32.c
  ${zlib_source_SOURCE_DIR}/inffast.c ${zlib_source_SOURCE_DIR}/inflate.c
  ${zlib_source_SOURCE_DIR}/inftrees.c ${zlib_source_SOURCE_DIR}/zutil.c)
target_include_directories(ehud_inflate PUBLIC ${zlib_source_SOURCE_DIR})
add_library(runtime_resources resources/resource_data.cpp)
target_include_directories(runtime_resources PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(runtime_resources PRIVATE ehud_inflate)
add_library(watch_scene scene/watch_scene.cpp)
target_include_directories(watch_scene PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(projected_editor platform/projected_editor.cpp platform/composition_probe.cpp)
add_library(rendered_cursor platform/rendered_cursor.cpp)
target_include_directories(rendered_cursor PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(rendered_cursor PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(rendered_cursor PUBLIC runtime_resources windowscodecs bcrypt gdi32 user32 ole32 windowsapp uuid)
target_include_directories(projected_editor PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(projected_editor PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(projected_editor PUBLIC d2d1 dwrite ole32 oleaut32 user32 uuid runtimeobject dcomp windowsapp uiautomationcore)
add_library(source_draw render/source_draw.cpp)
target_include_directories(source_draw PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(source_draw PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(source_draw PUBLIC watch_scene runtime_resources d3d11 d2d1 dwrite d3dcompiler windowsapp)
add_executable(EndfieldHUDWindows WIN32 app/main.cpp render/native_renderer.cpp app/app.manifest)
target_compile_definitions(EndfieldHUDWindows PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(EndfieldHUDWindows PRIVATE source_draw projected_editor rendered_cursor d3d11 dxgi dcomp d2d1 dwrite d3dcompiler shell32 ole32 psapi shcore)
if(MSVC)
  foreach(target watch_scene projected_editor rendered_cursor source_draw runtime_resources EndfieldHUDWindows)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(source_draw_tests tests/source_draw_tests.cpp)
  target_link_libraries(source_draw_tests PRIVATE source_draw dxgi runtimeobject ole32 uuid)
  add_test(NAME source_material_gpu_contracts COMMAND source_draw_tests "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/Resources/WatchSource")
  add_executable(rendered_cursor_tests tests/rendered_cursor_tests.cpp)
  target_link_libraries(rendered_cursor_tests PRIVATE rendered_cursor)
  add_test(NAME original_cursor_contracts COMMAND rendered_cursor_tests "${CMAKE_CURRENT_SOURCE_DIR}/../Resources/WatchSource/Cursor")
  add_executable(resource_tests tests/resource_tests.cpp)
  target_link_libraries(resource_tests PRIVATE runtime_resources)
  add_test(NAME native_resource_decoder COMMAND resource_tests)
  add_executable(scene_tests tests/scene_tests.cpp)
  target_link_libraries(scene_tests PRIVATE watch_scene)
  add_test(NAME scene_contracts COMMAND scene_tests)
  add_test(NAME actual_source_contracts COMMAND scene_tests "${CMAKE_CURRENT_SOURCE_DIR}/../Resources/WatchSource/Scene")
  add_executable(editor_tests tests/editor_tests.cpp)
  target_link_libraries(editor_tests PRIVATE projected_editor)
  if(MSVC)
    target_compile_options(scene_tests PRIVATE /utf-8)
    target_compile_options(editor_tests PRIVATE /utf-8)
  endif()
  add_test(NAME projected_editor_contracts COMMAND editor_tests)
  add_executable(platform_probe tests/composition_probe_main.cpp)
  target_link_libraries(platform_probe PRIVATE projected_editor d3d11)
  if(MSVC)
    target_compile_options(platform_probe PRIVATE /utf-8)
  endif()
endif()
