cmake_minimum_required(VERSION 3.25)
project(EndfieldHUDWindows VERSION 0.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# Clean port of the current macOS application. The rejected Windows preview is
# not a dependency, build input, resource source or packageable application.
add_library(hud_scene core/scene.cpp)
target_include_directories(hud_scene PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(hud_motion core/motion.cpp)
target_include_directories(hud_motion PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(hud_data core/data/json.cpp core/data/file_io.cpp core/data/data_store.cpp core/data/notes_store.cpp)
target_include_directories(hud_data PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(hud_services native/system_services.cpp)
target_include_directories(hud_services PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(hud_packet core/shell_packet.cpp)
target_include_directories(hud_packet PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(hud_packet PUBLIC hud_data hud_scene)
add_library(hud_source_motion core/source_animation.cpp core/source_camera.cpp)
target_include_directories(hud_source_motion PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(hud_source_motion PUBLIC hud_data hud_motion hud_scene)
if(WIN32)
  target_compile_definitions(hud_data PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(hud_data PRIVATE winsqlite3)
  target_compile_definitions(hud_services PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 UNICODE _UNICODE)
  target_link_libraries(hud_services PRIVATE user32 shell32 ole32 uuid)
  target_link_libraries(hud_packet PRIVATE bcrypt)
else()
  find_package(SQLite3 REQUIRED)
  target_link_libraries(hud_data PRIVATE SQLite::SQLite3)
endif()
foreach(target hud_scene hud_motion hud_data hud_services hud_packet hud_source_motion)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()

if(WIN32)
  add_executable(compile_source_shaders tools/compile_source_shaders.cpp)
  target_compile_definitions(compile_source_shaders PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 UNICODE _UNICODE)
  target_link_libraries(compile_source_shaders PRIVATE hud_data d3dcompiler bcrypt)
  if(MSVC)
    target_compile_options(compile_source_shaders PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_library(hud_native native/renderer.cpp native/source_graphics.cpp)
  target_include_directories(hud_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(hud_native PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 UNICODE _UNICODE)
  target_link_libraries(hud_native PUBLIC hud_scene PRIVATE d3d11 dxgi dcomp d3dcompiler)
  add_executable(replay_source_packet tools/replay_source_packet.cpp)
  target_compile_definitions(replay_source_packet PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 UNICODE _UNICODE)
  target_link_libraries(replay_source_packet PRIVATE hud_native hud_packet user32 ole32)
  if(MSVC)
    target_compile_options(replay_source_packet PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  if(MSVC)
    target_compile_options(hud_native PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    foreach(test shader_translation source_frame_comparison)
      add_test(NAME ${test}_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${test}.py")
    endforeach()
    add_test(NAME packet_export_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_shell_packet_export.py" --self-test)
  endif()
  add_executable(source_camera_tests tests/source_camera_tests.cpp)
  target_link_libraries(source_camera_tests PRIVATE hud_source_motion)
  add_executable(source_animation_tests tests/source_animation_tests.cpp)
  target_link_libraries(source_animation_tests PRIVATE hud_source_motion hud_packet)
  add_test(NAME source_animation_contracts COMMAND source_animation_tests)
  add_test(NAME source_camera_contracts COMMAND source_camera_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/source-camera.json")
  set_tests_properties(source_camera_contracts source_animation_contracts PROPERTIES TIMEOUT 60)
  if(MSVC)
    target_compile_options(source_camera_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_compile_options(source_animation_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  foreach(name scene motion data shell_packet)
    add_executable(${name}_tests tests/${name}_tests.cpp)
    if(name STREQUAL "shell_packet")
      target_link_libraries(${name}_tests PRIVATE hud_packet)
    else()
      target_link_libraries(${name}_tests PRIVATE hud_${name})
    endif()
    if(MSVC)
      target_compile_options(${name}_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
    endif()
    add_test(NAME ${name}_contracts COMMAND ${name}_tests)
    set_tests_properties(${name}_contracts PROPERTIES TIMEOUT 60)
  endforeach()
  add_executable(system_services_tests tests/system_services_tests.cpp)
  target_link_libraries(system_services_tests PRIVATE hud_services)
  if(MSVC)
    target_compile_options(system_services_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME system_services_contracts COMMAND system_services_tests)
  set_tests_properties(system_services_contracts PROPERTIES TIMEOUT 60)
  if(WIN32)
    add_executable(native_renderer_tests tests/native_renderer_tests.cpp)
    target_compile_definitions(native_renderer_tests PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 UNICODE _UNICODE)
    target_link_libraries(native_renderer_tests PRIVATE hud_native user32 ole32)
    if(MSVC)
      target_compile_options(native_renderer_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
    endif()
    add_test(NAME native_renderer_contracts COMMAND native_renderer_tests "${CMAKE_CURRENT_SOURCE_DIR}/native/hud.hlsl")
    set_tests_properties(native_renderer_contracts PROPERTIES TIMEOUT 120)
    add_executable(source_graphics_tests tests/source_graphics_tests.cpp)
    target_link_libraries(source_graphics_tests PRIVATE hud_native d3dcompiler)
    if(MSVC)
      target_compile_options(source_graphics_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
    endif()
    add_test(NAME source_graphics_contracts COMMAND source_graphics_tests "${CMAKE_CURRENT_SOURCE_DIR}/native/hud.hlsl")
    set_tests_properties(source_graphics_contracts PROPERTIES TIMEOUT 120)
  endif()
endif()
